        function updateMOQLabel() {
            var selectedProduct = document.getElementById("productSelect").value;
            var unitLabel = document.getElementById("unitLabel");
            var moqValue = document.getElementById("moqValue");
            var volumeInput = document.getElementById("volumeInput");
            switch (selectedProduct) {
                case "almond":
                    unitLabel.innerText = "kg";
                    moqValue.innerText = "30 kg";
                    volumeInput.placeholder = "Enter target weight in kg";
                break;
                 case "Saffron":
                    unitLabel.innerText = "gram";
                    moqValue.innerText = "25 g";
                    volumeInput.placeholder = "Enter target weight in gram";
                break;
                 case "Silver Leaves":
                    unitLabel.innerText = "PKT";
                    moqValue.innerText = "10 PKT";
                    volumeInput.placeholder = "Enter target weight in PKT";
                break;
                case "cashews":
                case "Pistachio":
                    unitLabel.innerText = "kg";
                    moqValue.innerText = "10 kg";
                    volumeInput.placeholder = "Enter target weight in kg";
                break;
                default:
                    unitLabel.innerText = "kg";
                    moqValue.innerText = "0 kg";
                    volumeInput.placeholder = "Enter target weight";
                break;
            }

            validateQuantity();
        }

        function validateQuantity() {
            var selectedProduct = document.getElementById("productSelect").value;
            var quantity = parseFloat(document.getElementById("volumeInput").value);
            var warningElement = document.getElementById("moqWarning");

            if (isNaN(quantity)) {
                warningElement.classList.add("hidden");
                return;
            }

            var limit = (selectedProduct === "almonds") ? 30 : (selectedProduct === "Saffron") ? 25 : (selectedProduct === "Silver Leaves") ? 10 : (selectedProduct === "cashews" || selectedProduct === "Pistachio") ? 10 : 0; 

            if (quantity < limit) {
                warningElement.classList.remove("hidden");
            } else {
                warningElement.classList.add("hidden");
            }
        }
    </script>
